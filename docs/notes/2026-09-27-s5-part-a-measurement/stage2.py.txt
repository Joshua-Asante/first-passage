"""Stage 2 / PA-5 for S5 Part A (r2 §12.5, §9 PA-5; TEST_ONLY): prints stage2.json.

Pure arithmetic over retained evidence; it dispatches nothing and downloads nothing.
The `.py.txt` suffix keeps it out of the import-boundary gate, like the harness.

    python stage2.py.txt --artifact-zip <run 36766144433's qualification-s2-supervision zip> \
        --run-api <gh api .../actions/runs/36766144433> \
        --artifacts-api <gh api .../actions/runs/36766144433/artifacts> \
        --evidence-output <s2_run_evidence.py stdout for that run> \
        --stage1c <stage1c/36647434808-combined.json> \
        --stage1c-job-b <stage1c/36647434808-b-attempt1/record.json> > stage2.json

Run binding: the artifact's contents carry no GitHub run id, so the inputs are bound to
run 36766144433 through GitHub's own records. The run listing names the run and head, the
artifact listing ties the artifact id to that run and gives its SHA-256 digest, and the zip
read here must hash to that digest. Every input is then read from inside that zip, and each
is pinned below. The release, profile and policy identities come from the run's journal,
and each is checked against the bytes that carry it.

Operator rulings 2026-09-30 (ledger, "Operator rulings, 2026-09-30", item 2): for
TEST_ONLY, PA-5 is CPU-only, and `memory_peak_bytes` is an upper bound on the shared
footprint only. It is recorded, never attributed to Part A, and never a PA-5 input.

PA2B_FACTOR = Fraction(3, 2)


def digest(data):
    return hashlib.sha256(data).hexdigest()


def pinned(name, data):
    assert digest(data) == PINS[name], "%s does not match its pin" % name
    return data


def exact(value):
    """A recorded float as the exact decimal it prints as (the harness writes repr)."""
    return Fraction(repr(value)) if isinstance(value, float) else Fraction(value)


def ceil10(value):
    return 10 * math.ceil(value / 10)


def num(value, places=9):
    return round(float(value), places)


def canonical_sha256(value):
    return digest(json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode())


def ceilings(c_hat, w_hat, p_hat):
    """PA-1 and PA-2 after Stage 1c (r2 §9), rounded up to whole 10 s."""
    b = max(M_C * c_hat, PA2B_FACTOR * p_hat)
    cpu = ceil10(max(CPU_FLOOR_S, b + O_S))
    wall_raw = M_W * (w_hat + L_S)
    wall = ceil10(max(WALL_FLOOR_S, wall_raw))
    return {"B_s": num(b), "B_plus_O_s": num(b + O_S), "cpu_s": cpu, "wall_raw_s": num(wall_raw), "wall_s": wall,
            "cpu_headroom_s": num(CPU_FLOOR_S - (b + O_S)), "wall_headroom_s": num(WALL_FLOOR_S - wall_raw),
            "m_w_ge_m_c_over_m_c_minus_1": M_W >= M_C / (M_C - 1)}


def bind_run(args):
    """The run, its artifact and the zip's bytes, from GitHub's own records."""
    run = json.loads(pinned("run_api.json", Path(args.run_api).read_bytes()))
    listing = json.loads(pinned("artifacts_api.json", Path(args.artifacts_api).read_bytes()))
    assert run["id"] == RUN_ID and run["head_sha"] == HEAD, "run listing is not the claimed run"
    assert run["status"] == "completed" and run["conclusion"] == "success"
    matches = [a for a in listing["artifacts"] if a["name"] == ARTIFACT_NAME]
    assert len(matches) == 1, matches
    artifact = matches[0]
    assert artifact["workflow_run"]["id"] == RUN_ID and artifact["workflow_run"]["head_sha"] == HEAD
    zip_bytes = pinned("artifact_zip", Path(args.artifact_zip).read_bytes())
    assert artifact["digest"] == "sha256:" + digest(zip_bytes), "zip differs from the run's artifact digest"
    return run, artifact, zipfile.ZipFile(io.BytesIO(zip_bytes))


def identities(journal_bytes, snapshot):
    """The measured release, profile and policy, each checked against the bytes that carry it."""
    with tempfile.TemporaryDirectory() as scratch:
        path = Path(scratch) / "journal.sqlite"
        path.write_bytes(journal_bytes)
        db = sqlite3.connect("file:%s?mode=ro" % path.as_posix(), uri=True)
        try:
            receipt = json.loads(db.execute("SELECT receipt_bytes FROM full_campaigns WHERE attempt_id=?",
                                            (ATTEMPT,)).fetchone()[0])
            objects = {role: (sha, bytes(body)) for role, sha, body in db.execute(
                "SELECT role, sha256, body FROM full_campaign_objects WHERE attempt_id=?", (ATTEMPT,))}
        finally:
            db.close()
    for role, (sha, body) in objects.items():
        assert digest(body) == sha, "journal object %s does not hash to its sha256" % role
    release_sha, release_body = objects["context_execution_release"]
    policy_sha, policy_body = objects["context_qualification_policy"]
    release, policy = json.loads(release_body), json.loads(policy_body)
    assert receipt["attempt_id"] == ATTEMPT
    assert receipt["execution_release_sha256"] == release_sha
    assert receipt["policy_sha256"] == policy_sha == release["qualification_policy_sha256"]
    assert receipt["profile_sha256"] == release["profile_sha256"] == canonical_sha256(release["profile"])
    assert snapshot["profile"]["installed_profile_sha256"] == release["profile_sha256"]
    assert receipt["budget_profile_sha256"] == canonical_sha256(release["campaign_budget_profile"])
    assert release["campaign_budget_profile"]["phases"]["PART_A"] == snapshot["profile"]["phases"]["PART_A"]
    return {
        "source": "%s: the attempt's admission receipt (full_campaigns.receipt_bytes) and its retained "
                  "context objects (full_campaign_objects); every digest below is recomputed from the bytes" % JOURNAL,
        "release": {"schema": release["schema"], "sha256": release_sha, "authority_class": release["authority_class"],
                    "capability": release["capability"], "dispatch_checkpoints": release["dispatch_checkpoints"]},
        "profile": {"schema": release["profile"]["schema"], "sha256": release["profile_sha256"],
                    "basis": "sha256 of the release's profile in canonical JSON; equals the receipt's profile_sha256 "
                             "and the budget snapshot's installed_profile_sha256"},
        "policy": {"schema": policy.get("schema"), "sha256": policy_sha,
                   "basis": "the retained qualification policy object; equals the receipt's policy_sha256 and the "
                            "release's qualification_policy_sha256"},
        "budget_profile_sha256": receipt["budget_profile_sha256"],
        "budget_sha256": receipt["budget_sha256"],
        "receipt_schema": receipt["schema"],
        "receipt_authority_class": receipt["authority_class"],
    }
